function Get-Section([string]$content, [string]$heading) {
    $pattern = "(?ms)^# $([regex]::Escape($heading))\s*\r?\n(.*?)(?=^# |\z)"
    $match = [regex]::Match($content, $pattern)
    if (-not $match.Success) { return '' }
    return $match.Groups[1].Value
}
